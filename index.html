// Optimized Phonics Phonetic Mapping for Web Speech Synthesis const PHONICS_MAP = { 'A': 'ah', 'B': 'buh', 'C': 'kuh', 'D': 'duh', 'E': 'ehh', // Pronounces the short 'e' sound as in 'egg' 'F': 'ff', 'G': 'guh', 'H': 'hha', // Forces breathy 'h' sound instead of saying 'aitch' 'I': 'ih', 'J': 'juh', 'K': 'kuh', 'L': 'll', 'M': 'mm', 'N': 'nn', 'O': 'aw', 'P': 'puh', 'Q': 'kwuh', 'R': 'err', 'S': 'ss', 'T': 'tuh', 'U': 'uh', 'V': 'vuh', 'W': 'wuh', 'X': 'eks', 'Y': 'yuh', 'Z': 'zz' }; function speakLetter(letter) { letter = letter.toUpperCase(); if (audioMode === 'phonics' && PHONICS_MAP[letter]) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(PHONICS_MAP[letter]); utterance.rate = 0.8; // Slightly slower rate makes phonics crisp and clear utterance.pitch = 1.0; utterance.lang = 'en-US'; window.speechSynthesis.speak(utterance); } } else { speakText(letter); } }
